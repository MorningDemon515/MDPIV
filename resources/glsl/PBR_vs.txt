#version 330 core

const int MAX_BONES = 100;

layout (location = 0) in vec3 aPos;
layout (location = 1) in vec3 aNormal;
layout (location = 2) in vec2 aTexCoords;
layout (location = 3) in vec3 aTangent;
layout (location = 4) in vec3 aBitangent;
layout (location = 5) in ivec4 aBoneIDs;
layout (location = 6) in vec4 aWeights;

uniform mat4 model;
uniform mat4 view;
uniform mat4 projection;

uniform mat4 nm;

out vec3 Normal;
out vec3 FragPos;
out vec2 TexCoords;
out mat3 TBN;

uniform mat4 boneTransforms[MAX_BONES];

void main()
{
    mat4 boneTransform = mat4(0.0);
    for(int i = 0; i < 4; i++) {
        if(aBoneIDs[i] < 0) continue;
        if(aBoneIDs[i] >= MAX_BONES) {
            boneTransform = mat4(1.0);
            break;
        }
        boneTransform += boneTransforms[aBoneIDs[i]] * aWeights[i];
    }
    
    if(aBoneIDs[0] < 0) boneTransform = mat4(1.0);
    
    vec4 pos = boneTransform * vec4(aPos, 1.0);

    gl_Position = projection * view * model * pos;
    FragPos = vec3(model * pos);

    Normal = mat3(nm) * aNormal;
    TexCoords = aTexCoords;

    vec3 T = normalize( mat3(nm) * aTangent);
    vec3 B = normalize( mat3(nm) * aBitangent);
    vec3 N = normalize( mat3(nm) * aNormal);
    TBN = mat3(T, B, N);

}
