#version 330 core

out vec4 FragColor;

in vec3 Normal;
in vec3 FragPos;
in vec2 TexCoords;

uniform vec3 LightPos;

uniform vec3 ViewPos;

uniform samplerCube shadowMap;
uniform float far_plane;

const int PCF_SAMPLES = 16;
const vec3[16] PCF_OFFSETS = vec3[](
    vec3(0.0, 0.0, 1.0), vec3(0.0, 0.0, -1.0), vec3(0.0, 1.0, 0.0), vec3(0.0, -1.0, 0.0),
    vec3(1.0, 0.0, 0.0), vec3(-1.0, 0.0, 0.0), vec3(0.0, 1.0, 1.0), vec3(0.0, 1.0, -1.0),
    vec3(0.0, -1.0, 1.0), vec3(0.0, -1.0, -1.0), vec3(1.0, 0.0, 1.0), vec3(1.0, 0.0, -1.0),
    vec3(-1.0, 0.0, 1.0), vec3(-1.0, 0.0, -1.0), vec3(1.0, 1.0, 0.0), vec3(-1.0, 1.0, 0.0)
);

float ShadowCalculation(vec3 fragPos)
{
    vec3 fragToLight = fragPos - LightPos;
    float currentDepth = length(fragToLight);
    
    if(currentDepth > far_plane) return 1.0;
    
    float bias = max(0.05 * (1.0 - dot(normalize(fragToLight), normalize(Normal))), 0.005);
    
    float shadow = 0.0;
    float diskRadius = 0.02;
    
    for(int i = 0; i < PCF_SAMPLES; ++i)
    {
        vec3 sampleDir = normalize(fragToLight) + PCF_OFFSETS[i] * diskRadius;
        float closestDepth = texture(shadowMap, sampleDir).r * far_plane;
        
        shadow += (currentDepth - bias > closestDepth) ? 1.0 : 0.0;
    }
    
    shadow /= float(PCF_SAMPLES);
    
    return shadow;
}


void main()
{
   vec3 norm = normalize(Normal);

   vec3 LightDir = normalize(LightPos - FragPos);
   vec3 ViewDir = normalize(ViewPos - FragPos);
   vec3 HalfWayDir = normalize(LightDir + ViewDir);

   vec3 ambient = vec3(0.0);

   float diffuse = max(dot(norm , LightDir), 0.0);

   float specular = pow(max(dot(norm, HalfWayDir), 0.0), 32);

   float shadow = ShadowCalculation(FragPos);

   vec3 color = (ambient + (1.0 - shadow) * (diffuse + specular));

   color = color / (color + vec3(1.0));
   color = pow(color, vec3(1.0/2.2));
   FragColor = vec4(color, 1.0);
}
