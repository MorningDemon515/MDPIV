#version 330 core

out vec4 FragColor;

in vec2 TexCoords;
in vec3 Normal;
in vec3 FragPos;
in vec4 FragPosLightSpace;

uniform vec3 LightPos;

uniform sampler2D texture_diffuse0;
uniform sampler2D shadowMap;

uniform vec3 ViewPos;

vec3 ModelColor = texture(texture_diffuse0, TexCoords).rgb;

float ShadowCalculation(vec4 fragPosLightSpace, vec3 normal, vec3 lightDir)
{
    // 执行透视除法
    vec3 projCoords = fragPosLightSpace.xyz / fragPosLightSpace.w;
    // 变换到[0,1]的范围
    projCoords = projCoords * 0.5 + 0.5;
    
    // 如果超出远平面，则不在阴影中
    if(projCoords.z > 1.0)
        return 0.0;
        
    // 改进的阴影偏移计算
    float bias = 0.0;
    
    // 基于法线和光线方向的夹角计算偏移
    float cosTheta = clamp(dot(normal, lightDir), 0.0, 1.0);
    bias = 0.005 * tan(acos(cosTheta)); // 当表面与光线平行时增加偏移
    bias = clamp(bias, 0.001, 0.01);    // 限制偏移范围
    
    // 或者使用更简单但有效的方法
    // bias = max(0.05 * (1.0 - dot(normal, lightDir)), 0.005);
    
    // PCF柔化阴影
    float shadow = 0.0;
    vec2 texelSize = 1.0 / textureSize(shadowMap, 0);
    
    // 减小PCF采样范围（从3x3改为2x2或使用更小的偏移）
    for(int x = -1; x <= 1; ++x)
    {
        for(int y = -1; y <= 1; ++y)
        {
            float pcfDepth = texture(shadowMap, projCoords.xy + vec2(x, y) * texelSize * 0.5).r; 
            shadow += (projCoords.z - bias) > pcfDepth ? 1.0 : 0.0;        
        }    
    }
    shadow /= 9.0;
    
    // 处理超出阴影贴图边界的情况
    if(projCoords.z > 0.999)
        shadow = 0.0;
        
    return shadow;
}

void main()
{
    vec3 norm = normalize(Normal);
    vec3 LightDir = normalize(LightPos - FragPos);

    vec3 ViewDir = normalize(ViewPos - FragPos);
    vec3 ReflectDir = reflect(-LightDir, norm);

    vec3 ambient = vec3(0.0);

    float diffuse = max(dot(norm , LightDir), 0.0);

    float specular = pow(max(dot(ViewDir, ReflectDir), 0.0), 32);

    float shadow = ShadowCalculation(FragPosLightSpace, norm, LightDir);

    vec3 color = (ambient + (1.0 - shadow) * (diffuse + specular)) * ModelColor;

    color = color / (color + vec3(1.0));
    color = pow(color, vec3(1.0/2.2));
    FragColor = vec4(color, 1.0);
}
